const everypapers = require("@everypapers/sdk");

const client = new everypapers.Client(process.env.EVERYPAPERS_API_KEY);

async function generateInvoice() {
  const job = await client.generateDocument({
    templateId: "TMP_INVOICE_001",
    documentFormat: "pdf",
    records: [
      {
        fileName: "invoice_9942.pdf",
        fields: {
          invoiceNumber: "#INV-9942",
          customerName: "Acme Corporation",
          subtotal: 12000.00,
          tax: 1200.00,
          total: 13200.00
        }
      }
    ]
  });

  console.log("Document queued! Job ID:", job.jobId);
}

Have questions about EveryPapers? Here's everything you need to know.

EveryPapers merges your dynamic JSON payload with pre-configured reusable templates (uploaded via the dashboard or created via API). Our high-speed async engine compiles your variables, tables, and conditionals into a pristine PDF or DOCX file and delivers it via instant URL or webhook.
EveryPapers supports both PDF and DOCX output formats. You can specify the target format per API request using the document_format parameter.
Average document generation takes under 250ms for standard 1–5 page documents. For enterprise and bulk batch generation, our distributed async queues handle thousands of concurrent generation requests smoothly without rate limiting throttling.
